#include <iostream>
#include <string>
enum class Zone { Inner, Middle, Outer, Airport };
constexpr long long MIN_BALANCE_PAISE = 5000;
constexpr int PEAK_SURCHARGE_PERCENT = 20;
bool isPeak(int hour, int minute) {
   if (hour < 0 || hour > 23 || minute < 0 || minute > 59) return false;
   int m = hour * 60 + minute;
   return (m >= 8 * 60 && m < 10 * 60) ||
          (m >= 17 * 60 && m < 20 * 60);
}
long long baseFare(Zone zone) {
   switch (zone) {
       case Zone::Inner: return 1000;
       case Zone::Middle: return 2000;
       case Zone::Outer: return 3000;
       case Zone::Airport: return 6000;
   }
   return -1;
}
long long calculateFare(Zone zone, int hour, int minute) {
   long long base = baseFare(zone);
   if (base < 0 || hour < 0 || hour > 23 || minute < 0 || minute > 59)
       return -1;
   if (isPeak(hour, minute)) base += base * PEAK_SURCHARGE_PERCENT / 100;
   return base;
}
bool canTravel(long long balancePaise, long long farePaise) {
   return farePaise >= 0 &&
          balancePaise - farePaise >= MIN_BALANCE_PAISE;
}
bool recharge(long long& balancePaise, long long amountPaise) {
   if (amountPaise <= 0) return false;
   balancePaise += amountPaise;
   return true;
}
std::string money(long long paise) {
   return "Rs " + std::to_string(paise / 100) + "." +
          (paise % 100 < 10 ? "0" : "") + std::to_string(paise % 100);
}
int main() {
   struct Test { const char* name; Zone zone; int h; int m; long long expected; };
   Test tests[] = {
       {"Inner off-peak", Zone::Inner, 12, 0, 1000},
       {"Middle peak", Zone::Middle, 9, 0, 2400},
       {"Outer off-peak", Zone::Outer, 12, 0, 3000},
       {"08:00 boundary", Zone::Inner, 8, 0, 1200},
       {"10:00 boundary", Zone::Inner, 10, 0, 1000},
       {"17:00 boundary", Zone::Outer, 17, 0, 3600},
       {"20:00 boundary", Zone::Outer, 20, 0, 3000}
   };
   int passed = 0;
   for (const auto& t : tests) {
       long long actual = calculateFare(t.zone, t.h, t.m);
       bool ok = actual == t.expected;
       std::cout << t.name << ": expected " << t.expected
                 << ", actual " << actual << " paise -> "
                 << (ok ? "PASS" : "FAIL") << '\n';
       if (ok) ++passed;
   }
   std::cout << "Passed " << passed << "/" << (sizeof(tests)/sizeof(tests[0]))
             << " tests\n";
   long long balance = 6000;
   long long fare = 1000;
   std::cout << "Exact Rs 50 remaining accepted: "
             << std::boolalpha << canTravel(balance, fare) << '\n';
   std::cout << "Negative recharge accepted: "
             << recharge(balance, -500) << '\n';
}
