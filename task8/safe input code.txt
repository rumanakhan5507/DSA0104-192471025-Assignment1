#include <iomanip>
#include <iostream>
#include <limits>
int main() {
   int hour{};
   std::cout << "Enter hour (0-23): ";
   while (!(std::cin >> hour) || hour < 0 || hour > 23) {
       std::cout << "Invalid hour. Enter a whole number from 0 to 23: ";
       if (std::cin.fail()) {
           std::cin.clear();
           std::cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');
       }
   }
   long long farePaise = 1050;
   std::cout << "Fare: Rs " << farePaise / 100 << '.'
             << std::setw(2) << std::setfill('0') << farePaise % 100
             << std::setfill(' ') << '\n';
}
